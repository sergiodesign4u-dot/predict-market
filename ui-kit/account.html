The file

To change this component, edit components/account.css. To change a value it reads, edit the role in components/tokens.css.

components/account.css
/* components/account.css

   Component: account. Classes: .cta-bar, .flat.

   Reads: --bg-card-quiet, --bg-control, --bg-control-hover, --bg-pressed, --border-hairline, --color-action, --color-action-lit, --line-brass-soft, --text-on-brass, --text-primary
   Stand:  ui-kit/account.html
   Stands on: 3 ui-visual screens (how-it-works.html, my-profile.html, wallet.html)
   Colour goes through a role, geometry straight from a primitive. No em dash. */

/* THE BAR'S OWN SURFACE, and it is the one exception this step declared. The
   arrangement is in components/patterns/action-bar.css; a stone, a hairline
   along the top edge and two rounded top corners are not composition, and
   they are not the page plate either, so they stay with the component that
   draws them. */
.app-case .cta-bar{background:var(--bg-card-quiet);border:0;border-top:1px solid var(--border-hairline);border-radius:var(--radius-10) var(--radius-10) 0 0}
.app-case .cta-bar button{flex:1;border:1px solid var(--border-hairline);border-radius:var(--radius-10);background:var(--bg-control);color:var(--text-primary);font-family:var(--font-body);font-weight:var(--weight-semibold);font-size:var(--text-13);padding:var(--space-12);min-height:var(--control-44);cursor:pointer}
.app-case .cta-bar button:hover{border-color:var(--line-brass-soft);background:var(--bg-control-hover)}
.app-case .cta-bar>button:first-child,.app-case .cta-bar>a:first-child button{background:linear-gradient(135deg,var(--color-action),var(--color-action-lit));border:0;color:var(--text-on-brass);font-weight:var(--weight-bold);transition:box-shadow var(--dur-quick) ease}
/* Hover takes the LIT role to both stops, so the whole face of the plate comes up. It replaces
   filter:brightness(1.08), which was a raw number doing a colour job: a filter multiplies whatever
   is underneath it, so no theme can reach the value and the same 1.08 means two different things on
   the two grounds. Same three-step as the quick-amount chip in input.css. The glow is unchanged, and
   with the filter gone the rest rule above transitions the one property that still moves. */
.app-case .cta-bar>button:first-child:hover,.app-case .cta-bar>a:first-child button:hover{background:linear-gradient(135deg,var(--color-action-lit),var(--color-action-lit));border:0;color:var(--text-on-brass);box-shadow:0 6px 18px -8px var(--color-action)}
/* Press, the axis this bar never had: two of its three screens put the bar under a thumb, where
   there is no hover to answer with. The system's two mechanisms and no third. A quiet button settles
   onto --bg-pressed. The brass action turns its own gradient over, 135deg to 315deg, so the light
   falls to the bottom right the way a plate pushed IN does, and the hover glow goes, because the
   glow is the lift. Both rules mirror the hover selector they answer, so each sits at equal
   specificity to it and wins by source order while the pointer is down. Nothing carries a drop
   shadow at rest, so there is nothing else to flatten. */
.app-case .cta-bar button:active{background:var(--bg-pressed)}
.app-case .cta-bar>button:first-child:active,.app-case .cta-bar>a:first-child button:active{background:linear-gradient(315deg,var(--color-action),var(--color-action-lit));box-shadow:none}
.app-case .cta-bar.flat{border:0;background:none;border-radius:0}
